    // Core Custom Card Adder

    function addCustomCardToDatabase(...args) {

      let type, rawText, source;

      if (args.length >= 4) {

        type = args[1];

        rawText = args[2];

        source = args[3];

      } else {

        type = args[0];

        rawText = args[1];

        source = args[2];

      }

 

      const cleaned = cleanCardText(rawText);

      if (!isValidCardText(cleaned)) {

        showToast("⚠️ 輸入的卡片詞彙不符合要求（請勿輸入純數字或說明文字）。", "error");

        return false;

      }

 

      const pool = unifiedPool[type];

      if (!pool) return false;

      const exists = pool.some(c => c.text === cleaned);

      if (exists) {

        showToast("⚠️ 該卡片詞彙已經存在於卡池中囉！", "info");

        return false;

      }

 

      const newCard = { text: cleaned, source: source, isCustom: true };

      pool.push(newCard);

 

      saveCustomCardsToStorage();

      renderCardLists();

      updateCharts();

 

      const typeNames = { time: '時間', person: '人物', location: '地點', event: '事件' };

      showToast(`🎉 已成功新增${typeNames[type]}卡牌：「${cleaned}」！`, "success");

      return true;

    }

    // Multi-Format Excel Upload Handler (支援課文原句示範匯入)

    window.handleBatchFileUpload = async function(event) {

      const files = event.target.files;

      if (!files || files.length === 0) return;

 

      let importedCount = 0;

      let sentenceImported = 0;

 

      for (let fIdx = 0; fIdx < files.length; fIdx++) {

        const file = files[fIdx];

        try {

          const buffer = await file.arrayBuffer();

          const workbook = XLSX.read(new Uint8Array(buffer), { type: 'array' });

 

          workbook.SheetNames.forEach(sheetName => {

            const ws = workbook.Sheets[sheetName];

            const rows = XLSX.utils.sheet_to_json(ws, { header: 1 });

            if (!rows || rows.length === 0) return;

 

            let tIdx = -1, wIdx = -1, sIdx = -1, fullSentenceIdx = -1;

            let colMap = { time: -1, person: -1, location: -1, event: -1 };

 

            const header = rows[0] || [];

            header.forEach((c, i) => {

              const str = String(c || '').trim();

              if (str.includes("要素") || str.includes("類別") || str.includes("類型")) tIdx = i;

              else if (str.includes("詞語") || str.includes("卡牌") || str.includes("內容") || str.includes("詞彙") || str.includes("短語")) wIdx = i;

              else if (str.includes("出處") || str.includes("課次") || str.includes("來源") || str.includes("備註")) sIdx = i;

              else if (str.includes("原句") || str.includes("示範") || str.includes("例句") || str.includes("完整句")) fullSentenceIdx = i;

 

              if (str.includes("時")) colMap.time = i;

              else if (str.includes("人") || str.includes("角色") || str.includes("主角")) colMap.person = i;

              else if (str.includes("地") || str.includes("場")) colMap.location = i;

              else if (str.includes("事") || str.includes("動") || str.includes("情節")) colMap.event = i;

            });

 

            const detectType = (val) => {

              const s = String(val || '');

              if (s.includes("時")) return "time";

              if (s.includes("人") || s.includes("角色")) return "person";

              if (s.includes("地") || s.includes("場")) return "location";

              if (s.includes("事") || s.includes("動")) return "event";

              return null;

            };

 

            const tryAddCard = (cardType, rawText, src) => {

              const cleaned = cleanCardText(rawText);

              if (isValidCardText(cleaned) && unifiedPool[cardType]) {

                const exists = unifiedPool[cardType].some(c => c.text === cleaned);

                if (!exists) {

                  unifiedPool[cardType].push({ text: cleaned, source: src || "自訂匯入", isCustom: true });

                  importedCount++;

                }

              }

            };

 

            if (colMap.time !== -1 && colMap.person !== -1 && colMap.location !== -1 && colMap.event !== -1) {

              for (let r = 1; r < rows.length; r++) {

                const row = rows[r];

                if (!row) continue;

                const tW = cleanCardText(row[colMap.time]);

                const pW = cleanCardText(row[colMap.person]);

                const lW = cleanCardText(row[colMap.location]);

                const eW = cleanCardText(row[colMap.event]);

                const src = sIdx !== -1 && row[sIdx] ? String(row[sIdx]).trim() : sheetName;

                const fullTxt = (fullSentenceIdx !== -1 && row[fullSentenceIdx]) ? String(row[fullSentenceIdx]).trim() : `${tW}，${pW}${lW}${eW}。`;

 

                if (isValidCardText(tW) && isValidCardText(pW) && isValidCardText(lW) && isValidCardText(eW)) {

                  tryAddCard('time', tW, src);

                  tryAddCard('person', pW, src);

                  tryAddCard('location', lW, src);

                  tryAddCard('event', eW, src);

 

                  masterSentences.push({

                    fullText: fullTxt,

                    time: tW, person: pW, location: lW, event: eW,

                    source: src || "匯入課文示範"

                  });

                  sentenceImported++;

                }

              }

            } else if (tIdx !== -1 && wIdx !== -1) {

              for (let r = 1; r < rows.length; r++) {

                const row = rows[r];

                if (!row) continue;

                const pType = detectType(row[tIdx]);

                const src = sIdx !== -1 && row[sIdx] ? String(row[sIdx]).trim() : "自訂匯入";

                if (pType) tryAddCard(pType, row[wIdx], src);

              }

            }

          });

        } catch (err) {

          console.error("Error reading file:", err);

        }

      }

 

      saveCustomCardsToStorage();

      renderCardLists();

      updateCharts();

      generateQuizQuestion();

 

      if (importedCount > 0) {

        showToast(`🎉 成功匯入 ${importedCount} 張詞彙${sentenceImported > 0 ? `與 ${sentenceImported} 句課文示範原句` : ''}！`, "success");

      } else {

        showToast("⚠️ 未能抓取到有效詞彙，請確認檔案格式。", "error");

      }

    };
